{{/* 匹配 GitHub 风格的 [!NOTE], [!TIP], [!WARNING], [!IMPORTANT], [!CAUTION] */}} {{- $raw := .Text -}} {{- $type := "" -}} {{- $title := "" -}} {{- $icon := "" -}} {{- if strings.HasPrefix (strings.TrimLeft " \n\r\t" $raw) "

[!NOTE]" -}} {{- $type = "note" -}} {{- $title = "Note" -}} {{- $icon = "ℹ️" -}} {{- $raw = replaceRE `^

\[!NOTE\]\s*` "

" $raw 1 -}} {{- else if strings.HasPrefix (strings.TrimLeft " \n\r\t" $raw) "

[!TIP]" -}} {{- $type = "tip" -}} {{- $title = "Tip" -}} {{- $icon = "💡" -}} {{- $raw = replaceRE `^

\[!TIP\]\s*` "

" $raw 1 -}} {{- else if strings.HasPrefix (strings.TrimLeft " \n\r\t" $raw) "

[!WARNING]" -}} {{- $type = "warning" -}} {{- $title = "Warning" -}} {{- $icon = "⚠️" -}} {{- $raw = replaceRE `^

\[!WARNING\]\s*` "

" $raw 1 -}} {{- else if strings.HasPrefix (strings.TrimLeft " \n\r\t" $raw) "

[!IMPORTANT]" -}} {{- $type = "important" -}} {{- $title = "Important" -}} {{- $icon = "🔔" -}} {{- $raw = replaceRE `^

\[!IMPORTANT\]\s*` "

" $raw 1 -}} {{- else if strings.HasPrefix (strings.TrimLeft " \n\r\t" $raw) "

[!CAUTION]" -}} {{- $type = "caution" -}} {{- $title = "Caution" -}} {{- $icon = "🛑" -}} {{- $raw = replaceRE `^

\[!CAUTION\]\s*` "

" $raw 1 -}} {{- end -}} {{- if $type -}}

{{ $icon }} {{ $title }}
{{ $raw | safeHTML }}
{{- else -}}
{{ .Text | safeHTML }}
{{- end -}}
